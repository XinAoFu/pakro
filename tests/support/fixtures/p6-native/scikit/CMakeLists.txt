cmake_minimum_required(VERSION 3.18)
project(p6_scikit_native LANGUAGES CXX)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
file(WRITE "${CMAKE_BINARY_DIR}/probe.cpp" "int main() { return 0; }\n")
try_run(P6_RUN P6_COMPILED "${CMAKE_BINARY_DIR}/try-run" SOURCES "${CMAKE_BINARY_DIR}/probe.cpp")
if(NOT P6_COMPILED OR NOT P6_RUN EQUAL 0)
  message(FATAL_ERROR "the actual CMake try_run executable did not run")
endif()
Python_add_library(p6_native_demo MODULE WITH_SOABI src/demo.cpp)
install(TARGETS p6_native_demo DESTINATION .)
